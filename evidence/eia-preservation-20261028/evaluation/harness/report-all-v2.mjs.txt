import fs from 'node:fs';
import path from 'node:path';
import {fileURLToPath} from 'node:url';
const root=fileURLToPath(new URL('../',import.meta.url)),records=path.join(root,'records'),v2=path.join(records,'v2'),read=p=>JSON.parse(fs.readFileSync(p));
const baseline=read(path.join(v2,'rescored-baselines.json')),plan=read(path.join(v2,'reasoning-schedule.json'));
const median=a=>{const s=[...a].sort((a,b)=>a-b),i=Math.floor(s.length/2);return s.length?(s.length%2?s[i]:(s[i-1]+s[i])/2):null;};
const bounds=a=>({median:median(a),min:a.length?Math.min(...a):null,max:a.length?Math.max(...a):null});
const directory=path.join(v2,'reasoning-attempts'),attempts=fs.existsSync(directory)?fs.readdirSync(directory).filter(p=>p.endsWith('.result.json')).sort().map(p=>read(path.join(directory,p))):[];
const rows=attempts.map(a=>{
 const replay=a.replayFile?read(path.join(v2,a.replayFile)):null,final=replay?.final,publicResult=replay?.public;
 return {...a,downstream:{publicStatus:replay?.publicStatus,initialInterpreter:replay?.initial?.interpreter,initialFallback:replay?.initial?.fallback,initialScopeValidation:replay?.initial?.scopeValidation,final,sourceEvidencePreserved:replay?.evidencePreserved,externalReplayCalls:replay?.actualExternalCalls,variableCount:publicResult?.variables?.length||0,definitionCount:publicResult?.comparisonDefinitions?.length||0,userMessage:publicResult?.userMessage,warnings:publicResult?.userWarnings,candidates:publicResult?.variables||[],definitions:publicResult?.comparisonDefinitions||[],crossRouteFallback:replay?.pipeline?.diagnostics?.crossRouteFallback,pipelineError:replay?.pipelineError}};
});
function summary(effort){
 const all=rows.filter(a=>a.effort===effort),finals=[...new Map(all.filter(a=>!a.willRetry).map(a=>[a.scheduledId,a])).values()];
 const knownCost=all.reduce((s,a)=>s+(a.tokenCostUSD||0),0),usable=finals.filter(a=>a.grade?.usable&&a.httpStatus===200&&a.responseStatus==='completed').length;
 let repeatConsistency=0,passAgreement=0;
 for(let q=1;q<=10;q++){const pair=finals.filter(a=>a.queryId===q);if(pair.length!==2)continue;if(pair.every(a=>a.grade?.semantic?.gradable)&&JSON.stringify(pair[0].grade.semantic.signature)===JSON.stringify(pair[1].grade.semantic.signature))repeatConsistency++;if(pair.every(a=>a.grade)&&pair[0].grade.semantic.pass===pair[1].grade.semantic.pass)passAgreement++;}
 const tokens=all.reduce((s,a)=>({input:s.input+(a.usage?.input_tokens||0),cached:s.cached+(a.usage?.input_tokens_details?.cached_tokens||0),cacheWrite:s.cacheWrite+(a.usage?.input_tokens_details?.cache_write_tokens||0),output:s.output+(a.usage?.output_tokens||0),reasoning:s.reasoning+(a.usage?.output_tokens_details?.reasoning_tokens||0)}),{input:0,cached:0,cacheWrite:0,output:0,reasoning:0});
 const completedIds=new Set(finals.map(a=>a.scheduledId));
 const endToEnd=finals.map(a=>{const attempts=all.filter(x=>x.scheduledId===a.scheduledId);return Date.parse(a.completedAt)-Math.min(...attempts.map(x=>Date.parse(x.startedAt)));});
 return {label:'Luna-'+effort,model:'gpt-6-luna',effort,scheduled:20,completed:finals.length,pending:20-finals.length,actualAttempts:all.length,retries:all.filter(a=>a.attempt>1).length,correctedSemantic:finals.filter(a=>a.grade?.semantic?.pass).length,observedSemanticMistakes:finals.filter(a=>a.grade?.semantic?.status==='incorrect').length,ungradable:finals.filter(a=>a.grade?.semantic?.status==='ungradable').length,strictJSON:finals.filter(a=>a.grade?.strictJSON).length,shape:finals.filter(a=>a.grade?.shape).length,sourceSpans:finals.filter(a=>a.grade?.sourceSpans).length,contract:finals.filter(a=>a.grade?.contract?.pass).length,correctedUsable:usable,truncated:finals.filter(a=>a.truncated).length,httpFailures:all.filter(a=>a.httpStatus!==200).length,transportFailures:all.filter(a=>a.transportError).length,repeatConsistency,semanticPassAgreementPairs:passAgreement,latencyMs:bounds(finals.filter(a=>a.httpStatus===200).map(a=>a.elapsedMs)),endToEndScheduledLatencyMs:bounds(endToEnd),allAttemptLatencyMs:bounds(all.map(a=>a.elapsedMs)),tokens,costUSD:knownCost,unknownCostUpperBoundUSD:all.reduce((s,a)=>s+(a.unknownCostUpperBoundUSD||0),0),costPerUsable:usable?knownCost/usable:null,downstreamDisplayingResults:finals.filter(a=>a.downstream.variableCount||a.downstream.definitionCount).length,downstreamClarifications:finals.filter(a=>a.downstream.final?.needsClarification).length,perQuery:Array.from({length:10},(_,i)=>{const a=finals.filter(a=>a.queryId===i+1);return {queryId:i+1,completed:a.length,semantic:a.filter(a=>a.grade?.semantic.pass).length,contract:a.filter(a=>a.grade?.contract?.pass).length,usable:a.filter(a=>a.grade?.usable).length};})};
}
const models=[...baseline.models.map(m=>({...m,completed:20,pending:0,effort:'none',truncated:0,httpFailures:0,transportFailures:0,unknownCostUpperBoundUSD:0})),...plan.efforts.map(summary)];
const report={generatedAt:new Date().toISOString(),scheduledOverall:120,completedScheduled:models.reduce((s,m)=>s+m.completed,0),actualAttempts:40+attempts.length,knownTotalCostUSD:models.reduce((s,m)=>s+m.costUSD,0),unknownCostUpperBoundUSD:models.reduce((s,m)=>s+(m.unknownCostUpperBoundUSD||0),0),approvedCeilingUSD:read(path.join(v2,'budget-authorization.json')).totalCeilingUSD,maximumAuthorizedRunCostUSD:plan.budget.requiredTotalUSD,versionIdentity:plan.versionIdentity,models,baselineChangedVerdicts:baseline.changedVerdicts,newAttemptRows:rows,limitations:['All accuracy fractions use 20 scheduled responses per setting. Ungradable/truncated/failed responses are separate from observed semantic mistakes.','Strict original JSON only. App parser repairs and rules fallbacks do not upgrade model success.','Different 32768-token/300-second limits apply to reasoning runs vs original4096/30 none baselines; within four reasoning settings limits are identical.','Candidate display is not proof of appropriate variables, units or coverage. Inspect cached-retrieval warnings and known app issues independently.','Ten fixed queries cannot establish general accuracy or statistical equivalence.']};
fs.writeFileSync(path.join(v2,'comparison-all-settings.json'),JSON.stringify(report,null,2)+'\n');
console.log(JSON.stringify({completedScheduled:report.completedScheduled,actualAttempts:report.actualAttempts,knownTotalCostUSD:report.knownTotalCostUSD,models:models.map(({perQuery,...m})=>m)},null,2));
