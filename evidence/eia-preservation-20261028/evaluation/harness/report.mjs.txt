import fs from 'node:fs';
import {fileURLToPath} from 'node:url';
import path from 'node:path';
const root=fileURLToPath(new URL('../',import.meta.url)), records=path.join(root,'records');
const freeze=JSON.parse(fs.readFileSync(path.join(records,'frozen-evaluation.json')));
const official=JSON.parse(fs.readFileSync(path.join(records,'official-model-checks.json')));
const attemptDir=path.join(records,'attempts');
if(!fs.existsSync(attemptDir))throw Error('No paid comparison records yet');
const attempts=fs.readdirSync(attemptDir).filter(x=>x.endsWith('.result.json')).sort().map(x=>JSON.parse(fs.readFileSync(path.join(attemptDir,x))));
const median=a=>{const v=[...a].sort((x,y)=>x-y);if(!v.length)return null;const i=Math.floor(v.length/2);return v.length%2?v[i]:(v[i-1]+v[i])/2;};
function cost(a){const u=a.usage;if(!u)return null;const p=official.models[a.model].standardUSDPerMillion;const cached=u.input_tokens_details?.cached_tokens||0,written=u.input_tokens_details?.cache_write_tokens||0;return ((u.input_tokens-cached-written)*p.input+cached*p.cachedInput+written*p.cacheWrite+u.output_tokens*p.output)/1e6;}
function modelReport(model){
 const rows=attempts.filter(a=>a.model===model),finals=[...new Map(rows.map(r=>[r.scheduledId,r])).values()];
 const knownCost=rows.reduce((s,r)=>s+(cost(r)||0),0);const unknown=rows.filter(r=>!r.usage&&r.transportError);
 const usable=finals.filter(r=>r.httpStatus>=200&&r.httpStatus<300&&r.originalGrade?.semantic.pass&&r.originalGrade?.contract.pass);
 const generation=finals.filter(r=>r.httpStatus>=200&&r.httpStatus<300);let consistent=0,passAgreement=0;
 for(const c of freeze.cases){const pair=finals.filter(r=>r.queryId===c.id);if(pair.length!==2)continue;if(pair.every(r=>r.originalGrade?.semantic.signature)&&JSON.stringify(pair[0].originalGrade.semantic.signature)===JSON.stringify(pair[1].originalGrade.semantic.signature))consistent++;if(pair.every(r=>r.originalGrade)&&pair[0].originalGrade.semantic.pass===pair[1].originalGrade.semantic.pass)passAgreement++;}
 const downstream=finals.map(r=>{const p=r.replayFile?JSON.parse(fs.readFileSync(path.join(records,r.replayFile))):null;const candidates=p?.public?.variables||[],definitions=p?.public?.comparisonDefinitions||[];return {scheduledId:r.scheduledId,queryId:r.queryId,repeat:r.repeat,originalSemantic:r.originalGrade?.semantic.pass||false,originalContract:r.originalGrade?.contract.pass||false,originalErrors:r.originalGrade?.semantic.errors||r.gradingFailure||[],contractErrors:r.originalGrade?.contract.errors||[],interpreter:p?.initial?.interpreter,fieldFallback:p?.initial?.fallback,route:p?.final?.route?.family,geographies:p?.final?.geographies,pairs:p?.final?.pairs,frequency:p?.final?.frequency,explicit:p?.final?.frequencyExplicit,needsClarification:p?.final?.needsClarification,publicStatus:p?.publicStatus,candidateCount:candidates.length,definitionCount:definitions.length,pipelineError:p?.pipelineError,frequencyFallback:p?.pipeline?.diagnostics?.crossRouteFallback,warnings:p?.public?.userWarnings,units:[...new Set(candidates.map(c=>c.unit).concat(definitions.map(d=>d.definition?.unit)).filter(Boolean))],sourceScopePreserved:p?.evidencePreserved,finalMeaning:p?.final};});
 return {model,scheduledExpected:20,scheduledCompleted:finals.length,actualAttempts:rows.length,retries:rows.filter(r=>r.attempt>1).length,httpSuccesses:generation.length,semanticPasses:finals.filter(r=>r.originalGrade?.semantic.pass).length,strictJSONPasses:finals.filter(r=>r.originalGrade?.contract.json).length,shapePasses:finals.filter(r=>r.originalGrade?.contract.shape).length,spanPasses:finals.filter(r=>r.originalGrade?.contract.spans).length,contractPasses:finals.filter(r=>r.originalGrade?.contract.pass).length,usableOriginalInterpretations:usable.length,consistentQueryPairs:consistent,semanticPassAgreementPairs:passAgreement,latencyMs:{median:median(generation.map(r=>r.elapsedMs)),min:generation.length?Math.min(...generation.map(r=>r.elapsedMs)):null,max:generation.length?Math.max(...generation.map(r=>r.elapsedMs)):null},tokens:rows.reduce((s,r)=>({input:s.input+(r.usage?.input_tokens||0),cached:s.cached+(r.usage?.input_tokens_details?.cached_tokens||0),cacheWrite:s.cacheWrite+(r.usage?.input_tokens_details?.cache_write_tokens||0),output:s.output+(r.usage?.output_tokens||0)}),{input:0,cached:0,cacheWrite:0,output:0}),knownTokenCostUSD:knownCost,unknownBilledAttempts:unknown.length,unknownCostUpperBoundUSD:unknown.reduce((s,r)=>s+r.ceilingReservedUSD,0),costPerUsableUSD:usable.length?knownCost/usable.length:null,downstream};
}
const report={generatedAt:new Date().toISOString(),models:freeze.settings.models.map(modelReport),note:'Original meanings/contract scores are separate from downstream app repairs and metadata availability. All disagreements require human inspection against the frozen criteria. No additional model grading calls.'};
fs.writeFileSync(path.join(records,'comparison-report.json'),JSON.stringify(report,null,2)+'\n');
console.log(JSON.stringify(report.models.map(({downstream,...r})=>r),null,2));
