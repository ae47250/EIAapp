function safeName(name) {
  if (path.isAbsolute(name) || name.split('/').some(p => p === '..' || p === 'node_modules' || p === '.git' || /^\.env(?:\.|$)/.test(p))) throw new Error('Excluded or unsafe archive path: ' + name);
}
function scan(bytes, name, redact = false) {
  let text = bytes.toString('utf8');
  for (const [kind, pattern] of credentialPatterns) {
    // Reviewed, deliberately fake transport sentinel used only by the archived offline probe.
    const found = [...text.matchAll(pattern)].filter(match => !(kind === 'credential-literal' && /^OPENAI_API_KEY\s*=\s*'(?:injected-offline-fixture-not-a-key|offline-independent-injection)'$/.test(match[0])));
    if (!found.length) continue;
    if (!redact) throw new Error(`Secret-pattern review needed: ${name}; category=${kind}; count=${found.length}. Values withheld.`);
    text = text.replace(pattern, '[REDACTED_CREDENTIAL]');
    redactions.push({ file: name, kind, occurrences: found.length });
  }
  if (redact) {
    const count = (text.match(/\/Users\/[^/\s"'<>]+/g) || []).length;
    if (count) {
      text = text.replace(/\/Users\/[^/\s"'<>]+/g, '<HOME>');
      redactions.push({ file: name, kind: 'local-home-path', occurrences: count });
    }
    const pattern = /(?:\/private)?\/var\/folders\/[^\s"'<>]+\/T\/eia-audit-base-[A-Za-z0-9_-]+/g;
    const countBase = (text.match(pattern) || []).length;
    if (countBase) {
      text = text.replace(pattern, '<BASELINE>');
      redactions.push({ file: name, kind: 'temporary-baseline-path', occurrences: countBase });
    }
    return Buffer.from(text);
  }
  return bytes;
}
function put(name, data, category, source = null, redact = false) {
  safeName(name);
  const original = Buffer.isBuffer(data) ? data : Buffer.from(data);
  const bytes = /\.png$/.test(name) ? original : scan(original, name, redact);
  const destination = path.join(stage, name);
  fs.mkdirSync(path.dirname(destination), { recursive: true });
  fs.writeFileSync(destination, bytes);
  entries.push({ path: name, category, bytes: bytes.length, sha256: sha(bytes), ...(source ? { source, originalSha256: sha(original) } : {}) });
}
function copy(source, name, category, redact = false) {
  if (!fs.existsSync(source)) { missing.push({ item: name, reason: 'Source record unavailable at packaging time' }); return; }
  if (!fs.lstatSync(source).isFile()) throw new Error('Not a regular file: ' + name);
  put(name, fs.readFileSync(source), category, source.startsWith(wd + '/') ? source.slice(wd.length + 1) : path.basename(source), redact);
}
